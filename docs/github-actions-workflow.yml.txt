# LIVE FILE
# The Muse GitHub App cannot push files under .github/workflows/ (GitHub
# returns 403), so this workflow is managed by hand through the GitHub web
# UI. Keep this text identical to the live file at
# .github/workflows/pipeline.yml in the repository.
#
# What it does: runs the unit tests on every push; runs the full pipeline on
# a weekly schedule plus manual dispatch. The scheduled run publishes the
# refreshed dashboard exports + run report back to main (which auto-deploys
# the Vercel site) -- but only if the tests and every data-quality check
# pass first. Pushes made with GITHUB_TOKEN do not re-trigger workflows, so
# this cannot loop; Vercel still deploys because it listens to push webhooks.
---
name: pipeline

on:
  push:
    branches: [main]
  schedule:
    - cron: "0 12 * * 1" # Mondays 12:00 UTC: monthly source data, weekly is plenty
  workflow_dispatch:

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install -e ".[dev]"
      - run: pytest -q

  run-pipeline:
    if: github.event_name != 'push'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install -e ".[dev]"
      - run: pytest -q
      - run: python -m ttc_platform.run_pipeline
      - name: Fail loudly if a data-quality check failed
        run: |
          python - <<'PY'
          import json, sys
          report = json.load(open("data/reports/latest_run.json"))
          failed = [c["name"] for c in report["checks"] if not c["passed"]]
          print("failed checks:", failed)
          sys.exit(1 if failed else 0)
          PY
      - name: Publish refreshed exports to main
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add dashboard/public/data/dashboard.json \
                  dashboard/public/data/tables_by_line.json \
                  dashboard/public/data/daily.json \
                  dashboard/public/data/quarter.json \
                  docs/latest_run.md
          if git diff --quiet --cached; then
            echo "no data changes to publish"
          else
            git commit -m "data: scheduled pipeline refresh $(date -u +%F)"
            git push
          fi
